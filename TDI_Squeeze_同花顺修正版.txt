{说明: TDI挤压指标 - 基于RSI的动态交易指标结合布林带肯特纳通道挤压}

{=== 参数设置区域 ===}
RSI周期:=14;        {RSI计算周期}
波动带周期:=34;      {RSI波动带周期}
波动带倍数:=1.6185;  {波动带标准差倍数}
快速周期:=1;        {价格线周期}
慢速周期:=7;        {信号线周期}

{挤压检测参数}
BB周期:=21;         {布林带周期}
BB倍数:=2.0;        {布林带倍数}
KC周期:=21;         {肯特纳通道周期}
KC倍数:=1.2;        {肯特纳通道倍数}

{=== 核心计算模块 ===}
{RSI计算 - 使用标准公式}
LC:=REF(CLOSE,1);
RSI1:=SMA(MAX(CLOSE-LC,0),RSI周期,1)/SMA(ABS(CLOSE-LC),RSI周期,1)*100;

{RSI波动带计算}
RSI中轨:=MA(RSI1,波动带周期);
RSI波动:=STD(RSI1,波动带周期);
RSI上带:=RSI中轨+波动带倍数*RSI波动;
RSI下带:=RSI中轨-波动带倍数*RSI波动;

{信号线计算}
价格线:=MA(RSI1,快速周期);
信号线:=MA(RSI1,慢速周期);

{=== 挤压检测计算 ===}
{布林带}
BB中轨:=MA(CLOSE,BB周期);
BB标准差:=STD(CLOSE,BB周期);
BB上轨:=BB中轨+BB倍数*BB标准差;
BB下轨:=BB中轨-BB倍数*BB标准差;

{肯特纳通道}
KC中轨:=EMA(CLOSE,KC周期);
{真实波幅ATR计算}
TR:=MAX(MAX(HIGH-LOW,ABS(HIGH-REF(CLOSE,1))),ABS(LOW-REF(CLOSE,1)));
ATR:=MA(TR,KC周期);
KC上轨:=KC中轨+KC倍数*ATR;
KC下轨:=KC中轨-KC倍数*ATR;

{挤压状态}
挤压:=BB上轨<KC上轨 AND BB下轨>KC下轨;

{=== 主要输出线 ===}
TDI价格线:价格线,COLORWHITE,LINETHICK2;
TDI信号线:信号线,COLORYELLOW,LINETHICK2;
TDI上轨:RSI上带,COLORBLUE,LINETHICK1;
TDI下轨:RSI下带,COLORBLUE,LINETHICK1;

{=== 参考线 ===}
STICKLINE(1,80,80,0,0),COLORGRAY;
STICKLINE(1,20,20,0,0),COLORGRAY;
STICKLINE(1,50,50,0,0),COLORGRAY;

{=== 挤压信号显示 ===}
挤压信号:IF(挤压,10,0),COLORLIGREEN,LINETHICK3;

{=== 数值显示 ===}
DRAWTEXT(ISLASTBAR,RSI1,RSI1);

{=== 使用说明 ===}
{
指标说明:
1. 白色线(TDI价格线): RSI快速移动平均，反应价格变化
2. 黄色线(TDI信号线): RSI慢速移动平均，产生交易信号  
3. 蓝色上下轨: RSI动态超买超卖阈值
4. 挤压信号: 当布林带被肯特纳通道包裹时显示
}
